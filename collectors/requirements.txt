# Runtime dependencies for the shared collector image.
#
# One image serves price_collector, news_collector and sentiment_collector; the
# compose services differ only by `command:`. Pinned exactly so an upstream
# release cannot change behaviour between builds.
#
# The reference project also depends on pandas, used only to turn API responses
# into DataFrames and immediately back into JSON for Redis. Plain dicts do that
# without adding ~70 MB to the image, so it is deliberately absent.

redis==8.1.0
requests==2.34.2

# RSS parsing (news collector) and HTML cleanup of feed summaries.
feedparser==6.0.14
beautifulsoup4==4.15.0
lxml==6.1.3

# Retry with exponential backoff on every external API call. The reference has
# no retry: one transient 502 loses an entire collection cycle.
tenacity==9.1.4

# Reads .env when a collector is run directly on the host for testing. Inside
# Docker the variables are already present and this is a no-op.
python-dotenv==1.2.4
